# 1. Initialize NestJS with npm (simplest for 2-day MVP)
npm i -g @nestjs/cli
nest new sop-engine --package-manager npm
cd sop-engine

# 2. Install XState v5, Zod, and Prisma dependencies
npm install xstate zod @prisma/client
npm install --save-dev prisma vitest

# 3. Initialize Prisma configuration
npx prisma init --datasource-provider sqlite
